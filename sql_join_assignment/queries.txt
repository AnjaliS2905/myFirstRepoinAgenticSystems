CREATE TABLE users (
    user_id INTEGER PRIMARY KEY,
    name TEXT,
    signup_date DATE
);

CREATE TABLE orders (
    order_id INTEGER PRIMARY KEY,
    user_id INTEGER,
    amount INTEGER
);

CREATE TABLE activity (
    user_id INTEGER,
    login_count INTEGER
);

SELECT 
    u.user_id,
    u.name,
    SUM(o.amount) AS total_order_amount,
    a.login_count
FROM users u
LEFT JOIN orders o ON u.user_id = o.user_id
LEFT JOIN activity a ON u.user_id = a.user_id
GROUP BY u.user_id, u.name, a.login_count;

SELECT u.name, SUM(o.amount) AS total_order_amount
FROM users u
INNER JOIN orders o

    a.login_count

LEFT JOIN activity a ON u.user_id = a.user_id

SELECT u.name, AVG(o.amount) AS avg_order_amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id
GROUP BY u.name;
